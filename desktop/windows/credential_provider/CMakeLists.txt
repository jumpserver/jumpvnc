cmake_minimum_required(VERSION 3.16)
project(jumpvnc_credential_provider LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The JumpVNC credential provider requires 64-bit Windows")
endif()
enable_language(RC)
add_library(jumpvnc-credential-provider SHARED provider.cpp console_login.cpp provider.def provider.rc)
set_target_properties(jumpvnc-credential-provider PROPERTIES PREFIX "")
target_compile_features(jumpvnc-credential-provider PRIVATE cxx_std_17)
target_compile_definitions(jumpvnc-credential-provider PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(jumpvnc-credential-provider PRIVATE ole32 uuid advapi32 secur32 wtsapi32 shlwapi comctl32 user32)
add_executable(credential-provider-smoke EXCLUDE_FROM_ALL smoke.cpp console_login.cpp)
target_compile_features(credential-provider-smoke PRIVATE cxx_std_17)
target_link_libraries(credential-provider-smoke PRIVATE ole32 uuid advapi32 secur32 wtsapi32 user32 gdi32)
add_dependencies(credential-provider-smoke jumpvnc-credential-provider)
if(MINGW)
  target_link_options(jumpvnc-credential-provider PRIVATE -static)
  target_link_options(credential-provider-smoke PRIVATE -static)
endif()
