cmake_minimum_required(VERSION 3.16)
project(svnc_desktop LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The JumpVNC desktop backend requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(feature EXAMPLES TESTS SDL GTK QT LIBSSHTUNNEL OPENSSL GNUTLS GCRYPT SASL WEBSOCKETS TIGHTVNC_FILETRANSFER FFMPEG XCB PNG LZO)
  set(WITH_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(WITH_IPv6 OFF CACHE BOOL "" FORCE)
set(WITH_THREADS ON CACHE BOOL "" FORCE)
set(PREFER_WIN32THREADS ON CACHE BOOL "" FORCE)
set(WITH_JPEG ON CACHE BOOL "" FORCE)
set(WITH_ZLIB ON CACHE BOOL "" FORCE)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
set(SVNC_DEPENDENCIES "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Prepared dependency sources")
add_subdirectory("${SVNC_DEPENDENCIES}/libvncserver" libvncserver)
add_subdirectory(credential_provider)
include(h264.cmake)
if(NOT JPEG_FOUND OR NOT ZLIB_FOUND)
  message(FATAL_ERROR "JPEG and ZLIB required for a meaningful Tight encoding trial")
endif()
add_executable(svnc-desktop main.cpp h264_encoder.cpp h264_bridge.cpp cursor_bridge.cpp)
target_include_directories(svnc-desktop PRIVATE "${SVNC_DEPENDENCIES}/libvncserver/include" ${CMAKE_CURRENT_BINARY_DIR}/libvncserver/include ${JPEG_INCLUDE_DIR} ${ZLIB_INCLUDE_DIR})
target_link_libraries(svnc-desktop PRIVATE vncserver d3d11 dxgi user32 gdi32 iphlpapi mfplat mfuuid wmcodecdspuuid strmiids ole32 oleaut32)
add_executable(h264-smoke EXCLUDE_FROM_ALL h264_smoke.cpp h264_encoder.cpp)
target_compile_definitions(h264-smoke PRIVATE SVNC_H264_TESTING)
target_link_libraries(h264-smoke PRIVATE mfplat mfuuid wmcodecdspuuid strmiids ole32 oleaut32)
add_executable(input-smoke EXCLUDE_FROM_ALL input_smoke.cpp)
target_include_directories(input-smoke PRIVATE "${SVNC_DEPENDENCIES}/libvncserver/include" ${CMAKE_CURRENT_BINARY_DIR}/libvncserver/include)
target_link_libraries(input-smoke PRIVATE user32)
add_executable(cursor-smoke EXCLUDE_FROM_ALL cursor_smoke.cpp cursor_bridge.cpp)
target_include_directories(cursor-smoke PRIVATE "${SVNC_DEPENDENCIES}/libvncserver/include" ${CMAKE_CURRENT_BINARY_DIR}/libvncserver/include ${ZLIB_INCLUDE_DIR})
if(MINGW)
  target_link_options(svnc-desktop PRIVATE -municode -static)
  target_link_options(h264-smoke PRIVATE -static)
  target_link_options(input-smoke PRIVATE -static)
  target_link_options(cursor-smoke PRIVATE -static)
endif()

install(TARGETS svnc-desktop RUNTIME DESTINATION backends/windows)
install(FILES COPYING DESTINATION licenses/libvncserver)
