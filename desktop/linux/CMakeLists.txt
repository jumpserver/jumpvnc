cmake_minimum_required(VERSION 3.16)
project(svnc_desktop_linux LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This backend requires Linux X11")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11 xtst xfixes)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavcodec>=59 libavutil libswscale)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
# The system LibVNCServer does not implement our noVNC encoding-50 framing.
# Build a pristine, pinned dependency with the same RFB hooks as Windows.
set(SVNC_DEPENDENCIES "${CMAKE_CURRENT_SOURCE_DIR}" CACHE PATH "Prepared dependency sources")
if(NOT EXISTS "${SVNC_DEPENDENCIES}/libvncserver/CMakeLists.txt")
  message(FATAL_ERROR "Run desktop/linux/prepare-sources.sh first (or set SVNC_DEPENDENCIES)")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LIBVNCSERVER_INSTALL OFF CACHE BOOL "" FORCE)
foreach(feature EXAMPLES TESTS SDL GTK QT LIBSSHTUNNEL OPENSSL GNUTLS GCRYPT SASL WEBSOCKETS TIGHTVNC_FILETRANSFER FFMPEG XCB PNG LZO SYSTEMD)
  set(WITH_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(WITH_JPEG ON CACHE BOOL "" FORCE)
set(WITH_ZLIB ON CACHE BOOL "" FORCE)
set(WITH_THREADS ON CACHE BOOL "" FORCE)
add_subdirectory("${SVNC_DEPENDENCIES}/libvncserver" libvncserver)
include(h264.cmake)
add_executable(svnc-desktop main.cpp h264_encoder.cpp h264_bridge.cpp)
target_include_directories(svnc-desktop PRIVATE "${SVNC_DEPENDENCIES}/libvncserver/include" "${CMAKE_CURRENT_BINARY_DIR}/libvncserver/include")
target_link_libraries(svnc-desktop PRIVATE vncserver PkgConfig::X11 PkgConfig::FFMPEG)
target_compile_options(svnc-desktop PRIVATE -Wall -Wextra -Wpedantic)
add_executable(h264-smoke EXCLUDE_FROM_ALL h264_smoke.cpp h264_encoder.cpp)
target_compile_definitions(h264-smoke PRIVATE SVNC_H264_TESTING)
target_link_libraries(h264-smoke PRIVATE PkgConfig::FFMPEG)
target_compile_options(h264-smoke PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS svnc-desktop RUNTIME DESTINATION backends/linux)
install(FILES ../COPYING DESTINATION licenses/libvncserver)
