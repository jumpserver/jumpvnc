cmake_minimum_required(VERSION 3.16)
project(svnc_desktop_linux LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This backend requires Linux X11")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(VNC REQUIRED IMPORTED_TARGET libvncserver>=0.9.15)
pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11 xtst xfixes)
add_executable(svnc-desktop main.cpp)
target_link_libraries(svnc-desktop PRIVATE PkgConfig::VNC PkgConfig::X11)
target_compile_options(svnc-desktop PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS svnc-desktop RUNTIME DESTINATION backends/linux)
install(FILES ../COPYING DESTINATION licenses/libvncserver)
